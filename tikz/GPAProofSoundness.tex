\section*{(P2)}

\[
\Big[
\begin{tikzpicture}[axpic]
  \pic (m) at (0.500,0.500) {multiplication};
  \pic (g) at (1.750,0.500) {ge};
  \draw (-0.250,1.000) -- (m-in-0);
  \draw (-0.250,0.000) -- (m-in-1);
  \draw (m-out-0) to[out=0,in=180] (g-in-0);
\end{tikzpicture}
\Big]
=
\Big\{\big(\tbinom{x_1}{x_2},y\big)\mid x_1+x_2\ge y\Big\}
=
\Big\{\big(\tbinom{x_1}{x_2},y\big)\ \Big|\ \exists c,d.\ x_1+x_2\ge c+d \wedge c+d=y\Big\}
\]
\[
=
\Big\{\big(\tbinom{x_1}{x_2},y\big)\ \Big|\ \exists \hat c,\hat d.\ x_1\ge\hat c\wedge x_2\ge\hat d\wedge \hat c+\hat d=y\Big\}

\Big[

\section*{(P5), (P6)}

For $k>0$:
\[
\Big[
\begin{tikzpicture}[axpic]
  \pic[val=k] (s) at (0.500,0.500) {scalar};
  \pic (g) at (1.750,0.500) {ge};
  \draw (-0.250,0.500) -- (s-in-0);
  \draw (s-out-0) to[out=0,in=180] (g-in-0);
\end{tikzpicture}
\Big]
=
\{(x,y)\mid kx\ge y \wedge k>0\}
=
\Big\{(x,y)\ \Big|\ x\ge\frac{y}{k}\wedge k>0\Big\}
\]
\[
=
\{(x,y)\mid \exists c.\ x\ge c \wedge kc=y \wedge k>0\}
=
\Big[
\begin{tikzpicture}[axpic]
  \pic (g) at (0.500,0.500) {ge};
  \pic[val=k] (s) at (1.750,0.500) {scalar};
  \draw (-0.250,0.500) -- (g-in-0);
  \draw (g-out-0) to[out=0,in=180] (s-in-0);
\end{tikzpicture}
\Big]
\]

For $k<0$: the same computation, dividing by the negative $k$ flips the inequality, so
\[
\Big[

  \draw (s-out-0) to[out=0,in=180] (g-in-0);
\end{tikzpicture}
\Big]
=
\Big[
\begin{tikzpicture}[axpic]

  \draw (g-out-0) to[out=0,in=180] (s-in-0);
\end{tikzpicture}
\Big]
\]
